# Opt-in compiler-only Release optimization. Keep Debug/sanitizer gates unchanged.
option(NFS_THIN_LTO "Use checked, bounded ThinLTO for Release game targets" OFF)
set(NFS_THIN_LTO_JOBS 2 CACHE STRING "ThinLTO linker backend workers (1-3)")
# Enable only after all game targets have been defined.
include(CMakePushCheckState)
include(CheckCSourceCompiles)
include(CheckCXXSourceCompiles)
function(nfs_enable_thin_lto)
  if(NOT NFS_THIN_LTO)
    return()
  endif()
  if(NOT CMAKE_C_COMPILER_ID MATCHES "^(AppleClang|Clang)$" OR
     NOT CMAKE_CXX_COMPILER_ID MATCHES "^(AppleClang|Clang)$")
    message(FATAL_ERROR "NFS_THIN_LTO requires Clang or AppleClang")
  endif()
  if(NOT NFS_THIN_LTO_JOBS MATCHES "^[123]$")
    message(FATAL_ERROR "NFS_THIN_LTO_JOBS must be between 1 and 3")
  endif()
  if(APPLE)
    set(backend "-Wl,-mllvm,-threads=${NFS_THIN_LTO_JOBS}")
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(backend "-Wl,-plugin-opt,jobs=${NFS_THIN_LTO_JOBS}")
  else()
    message(FATAL_ERROR "ThinLTO linker options have not been validated for this platform")
  endif()
  cmake_push_check_state(RESET)
  set(CMAKE_REQUIRED_FLAGS "-flto=thin")
  set(CMAKE_REQUIRED_LINK_OPTIONS "-flto=thin;${backend}")
  unset(NFS_THIN_LTO_C_SUPPORTED CACHE)
  unset(NFS_THIN_LTO_CXX_SUPPORTED CACHE)
  check_c_source_compiles("int main(void){return 0;}" NFS_THIN_LTO_C_SUPPORTED)
  check_cxx_source_compiles("int main(){return 0;}" NFS_THIN_LTO_CXX_SUPPORTED)
  cmake_pop_check_state()
  if(NOT NFS_THIN_LTO_C_SUPPORTED OR NOT NFS_THIN_LTO_CXX_SUPPORTED)
    message(FATAL_ERROR "Compiler/linker cannot use ThinLTO with the bounded backend setting")
  endif()
  foreach(target IN LISTS ARGN)
    if(NOT TARGET "${target}")
      message(FATAL_ERROR "Missing explicit ThinLTO target: ${target}")
    endif()
    target_compile_options("${target}" PRIVATE "$<$<CONFIG:Release>:-flto=thin>")
    get_target_property(kind "${target}" TYPE)
    if(kind STREQUAL "EXECUTABLE" OR kind STREQUAL "SHARED_LIBRARY")
      target_link_options("${target}" PRIVATE "$<$<CONFIG:Release>:-flto=thin>" "$<$<CONFIG:Release>:${backend}>")
    endif()
  endforeach()
endfunction()

if(NFS_THIN_LTO)
  set(nfs_thin_lto_targets)

      NFSIISEEngineCheck NFSIISEGame engine_link_check engine_startup_check nfs_game_app)
    if(TARGET "${target}")
      list(APPEND nfs_thin_lto_targets "${target}")
    endif()
  endforeach()
  nfs_enable_thin_lto(${nfs_thin_lto_targets})
endif()
